// ============================================================ // EST Machine Booking — Google Apps Script Backend v3 // Paste into script.google.com and deploy as Web App // Execute as: Me | Who has access: Anyone // // Files created in your Google Drive: // EST_Bookings_Data.json — live booking records // EST_Settings_Data.json — hospitals / engineers / cal dates // EST_Usage_Log — Google Sheet usage record // // ► FIRST TIME SETUP: Run the setup() function manually from // the Apps Script editor to create the sheet immediately. // Click the function dropdown → select "setup" → click Run. // ============================================================ const BOOKINGS_FILE = 'EST_Bookings_Data.json'; const SETTINGS_FILE = 'EST_Settings_Data.json'; const SHEET_NAME = 'EST_Usage_Log'; const HEADERS = [ 'Recorded At', 'Action', 'Booking ID', 'Machine', 'Engineer', 'Purpose', 'Hospital', 'Date Out', 'Slot Out', 'Date Return', 'Slot Return', 'Notes' ]; const MACHINE_NAMES = { 'A': 'EST 612 (A)', 'C': 'EST 612 (C)', 'A2': 'EST 615 (A)', 'B2': 'EST 615 (B)' }; // ============================================================ // ► RUN THIS MANUALLY ONCE FROM THE SCRIPT EDITOR // to create EST_Usage_Log in your Google Drive immediately. // ============================================================ function setup() { const sheet = getOrCreateSheet(); const url = sheet.getParent().getUrl(); Logger.log('✓ EST_Usage_Log sheet is ready: ' + url); Logger.log('Go to Google Drive → My Drive → EST_Usage_Log, or open: ' + url); } // ── Main handler ── function doGet(e) { const action = e.parameter.action || 'load'; try { if (action === 'load') { return jsonOk({ bookings: getFile(BOOKINGS_FILE), settings: getFile(SETTINGS_FILE) }); } if (action === 'add') { const booking = JSON.parse(decodeURIComponent(e.parameter.booking)); const data = getFile(BOOKINGS_FILE); data.unshift(booking); saveFile(BOOKINGS_FILE, data); logToSheet('BOOKED', booking); return jsonOk({ success: true }); } if (action === 'delete') { const id = e.parameter.id; const data = getFile(BOOKINGS_FILE); const target = data.find(b => b.id === id); const updated = data.filter(b => b.id !== id); saveFile(BOOKINGS_FILE, updated); if (target) logToSheet('RETURNED', target); return jsonOk({ success: true }); } if (action === 'saveSettings') { const settings = JSON.parse(decodeURIComponent(e.parameter.settings)); saveFile(SETTINGS_FILE, settings); return jsonOk({ success: true }); } return jsonError('Unknown action: ' + action); } catch (err) { return jsonError(err.toString()); } } // ── Write a row to the Google Sheet ── function logToSheet(action, booking) { const sheet = getOrCreateSheet(); const now = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss'); sheet.appendRow([ now, action, booking.id || '', MACHINE_NAMES[booking.machineId] || booking.machineId || '', booking.engineer || '', booking.purpose || '', booking.hospital || '', booking.dateOut || '', booking.slotOut || '', booking.dateReturn || '', booking.slotReturn || '', booking.notes || '' ]); } // ── Get or create the usage log spreadsheet ── function getOrCreateSheet() { const files = DriveApp.getFilesByName(SHEET_NAME); if (files.hasNext()) { return SpreadsheetApp.open(files.next()).getSheets()[0]; } // Create new spreadsheet const ss = SpreadsheetApp.create(SHEET_NAME); const sheet = ss.getSheets()[0]; sheet.setName('Usage Log'); // Style header row const headerRange = sheet.getRange(1, 1, 1, HEADERS.length); headerRange.setValues([HEADERS]); headerRange.setFontWeight('bold'); headerRange.setBackground('#0f1117'); headerRange.setFontColor('#ffffff'); sheet.setFrozenRows(1); // Column widths const widths = [160, 90, 150, 110, 180, 110, 260, 100, 75, 100, 80, 200]; widths.forEach((w, i) => sheet.setColumnWidth(i + 1, w)); return sheet; } // ── JSON file helpers ── function getFile(name) { const files = DriveApp.getFilesByName(name); if (files.hasNext()) { const content = files.next().getBlob().getDataAsString(); try { return JSON.parse(content || (name === SETTINGS_FILE ? '{}' : '[]')); } catch(e) { return name === SETTINGS_FILE ? {} : []; } } return name === SETTINGS_FILE ? {} : []; } function saveFile(name, data) { const files = DriveApp.getFilesByName(name); if (files.hasNext()) { files.next().setContent(JSON.stringify(data)); } else { DriveApp.createFile(name, JSON.stringify(data), MimeType.PLAIN_TEXT); } } // ── Response helpers ── function jsonOk(payload) { return ContentService.createTextOutput(JSON.stringify(payload)).setMimeType(ContentService.MimeType.JSON); } function jsonError(msg) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: msg })).setMimeType(ContentService.MimeType.JSON); }
